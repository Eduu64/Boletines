# Puerto web del panel (panel: http://IP-DEL-SERVIDOR:PUERTO)
PUERTO=3000

# ── Acceso al panel: solo tu red local, nunca desde fuera ────────────────────
# Dos capas que trabajan juntas:
#  1) CORTAFUEGOS_ENTRADA=lan (más abajo): la app SOLO contesta a IPs privadas
#     (tu red local, redes Docker y loopback); una IP pública de internet
#     recibe un 403 inmediato. Esta capa es la que decide, y es la que manda.
#  2) PUERTO_PUBLICADO: en qué interfaz escucha Docker:
#     · 0.0.0.0 (por defecto) → escucha en todas; la capa 1 rechaza a internet.
#       Inmune a cambios de IP por DHCP y sin config extra: recomendado.
#     · <IP-local-del-servidor> (p. ej. 192.168.1.50) → Docker ni siquiera
#       abre el puerto en la interfaz pública (más estricto; con IP fija o
#       reserva DHCP, y el propio servidor deja de verlo por localhost).
#     Evita 0.0.0.0 solo si pones CORTAFUEGOS_ENTRADA=off a la vez.
PUERTO_PUBLICADO=0.0.0.0

# Techo de memoria del contenedor (la app suele rondar 50-80 MB)
LIMITE_MEMORIA=256m

# Programador automático de ingestas: 1 = activado, 0 = solo manual desde la web
ENABLE_SCHEDULER=1

# Horarios cron de ingesta (cada BOE se intenta a las 09:00 y se reintenta a las 11:35)
# Puedes añadir más: 0 9 * * *,35 11 * * *,0 17 * * *
HORARIO_INGESTA=0 9 * * *,35 11 * * *

# Zona horaria de los horarios anteriores
ZONA_HORARIA=Europe/Madrid

# Días hacia atrás que se reingestan al arrancar el contenedor si faltan (0-14).
# Así la ingesta diaria «queda pendiente» si el servidor estaba apagado a su hora:
# en cuanto la app arranca, rellena los días sin hacer (por defecto 7).
BACKFILL_DIAS=7

# ── Discord (opcional) ───────────────────────────────────────────────────────
# Modo recomendado: bot global (pega aquí el token y el canal por defecto).
# Modo alternativo: webhook por alerta (se pega en cada alerta desde la web).
DISCORD_BOT_TOKEN=
DISCORD_CANAL_ID=

# ── Seguridad (opcional, recomendado si el panel sale de tu red local) ──────
# Si defines ADMIN_TOKEN, toda escritura en la API y la consola del servidor
# exigirán la cabecera «x-admin-token: <valor>». Déjalo vacío si solo lo usas
# tú en tu red.
ADMIN_TOKEN=

# Permite mostrar el panel dentro de un iframe externo (p. ej. un panel de
# vista previa). Por defecto 0: la app no puede enmarcarse (anti-clickjacking).
# Actívalo solo si sabes que tu entorno lo necesita.
PERMITIR_EMBEBER=0

# ── Cortafuegos de salida (activado por defecto) ─────────────────────────────
# La app solo puede conectar hacia fuera a los portales de los 16 boletines y
# a Discord (discord.com/discordapp.com). Cualquier otro destino se bloquea
# antes de salir (incluidas IPs de tu red local 192.168.x.x — a propósito).
# CORTAFUEGOS_SALIDA=off lo desactiva. SALIDA_EXTRA añade dominios o IPs
# permitidos, separados por comas (p. ej. un proxy tuyo: SALIDA_EXTRA=192.168.1.2)
# ¿Blindaje extra a nivel de red? El «modo reforzado» (docker-compose.reforzado.yml)
# añade un proxy interno Squid con esta misma lista (proxy/dominios.list): la app
# vive en una red Docker sin internet y su única salida es ese proxy.
CORTAFUEGOS_SALIDA=lista
SALIDA_EXTRA=

# ── Cortafuegos de entrada: el panel solo se sirve a tu red (activado) ───────
# La app solo responde a IPs privadas (tu red local 192.168/10/172.16, redes
# Docker y loopback); una IP pública de internet recibe un 403 inmediato y se
# anota en la consola del servidor. CORTAFUEGOS_ENTRADA=off lo desactiva
# (no recomendado).
CORTAFUEGOS_ENTRADA=lan

# Zona horaria del sistema para los logs del contenedor
TZ=Europe/Madrid

# ── Solo para pruebas locales ────────────────────────────────────────────────
# Con PERMITIR_WEBHOOKS_LOCALES=1 se aceptan webhooks http://127.0.0.1/localhost
# para verificar el envío a Discord con un receptor propio (nunca en producción).
PERMITIR_WEBHOOKS_LOCALES=
